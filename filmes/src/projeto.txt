        <>

                type="range"
                min="1"

                {nota ? `Sua nota: ${nota} de 5` : 'Sem nota'}

        </>
    )

import { buscarFilme } from '../services/omdb.js'

export default function useFilmes(titulos) {
    const [filmes, setFilmes] = useState([])
    const [erro, setErro] = useState('')
    const [carregando, setCarregando] = useState(true)

    useEffect(() => {
        Promise.all(titulos.map((titulo) => buscarFilme(titulo)))
            .then(setFilmes)
            .catch(() => setErro('Não foi possível carregar os filmes. Confira a chave no arquivo .env'))
            .finally(() => setCarregando(false))
    }, [titulos])

    return { filmes, erro, carregando }
}import { useState } from 'react'

export default function useNotas() {
  const [notas, setNotas] = useState({})

  function avaliar(id, nota) {
    setNotas({ ...notas, [id]: nota })
  }

  const lista = Object.values(notas)
  const avaliados = lista.length
  const soma = lista.reduce((total, nota) => total + nota, 0)
  const media = avaliados > 0 ? soma / avaliados : 0

  return { notas, avaliar, avaliados, media }
}const chave = import.meta.env.VITE_OMDB_KEY

export function buscarFilme(titulo) {
    console.log(chave)
    const url = `https://www.omdbapi.com/?t=${encodeURIComponent(titulo)}&type=movie&apikey=${chave}`
    console.log(url)

    return fetch(url)
        .then((resposta) => resposta.json())
        .then((dados) => {
            if (dados.Response === 'False') {
                throw new Error(dados.Error)
            }

            return {
                id: dados.imdbID,
                titulo: dados.Title,
                ano: dados.Year,
                imagem: dados.Poster,
            }
        })
}
